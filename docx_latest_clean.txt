STEP 6 — Build the Booking page
Booking.jsx
import { useState } from "react";
function Booking() {
    const [name, setName] = useState("");
    const [tickets, setTickets] = useState("");
    return (
        <main className="booking-page">
            <h1>Book Your Ticket</h1>
            <div className="booking-form">
                <label htmlFor="name">Name</label>
                <input
                    id="name"
                    type="text"
                    value={name}
                    onChange={(e) => setName(e.target.value)}
                    placeholder="Enter your name"
                />
                <label htmlFor="tickets">Number of Tickets</label>
                <input
                    id="tickets"
                    type="number"
                    min="1"
                    value={tickets}
                    onChange={(e) => setTickets(e.target.value)}
                    placeholder="Enter number of tickets"
                />
            </div>
            <section className="booking-details">
                <h3>Booking Details</h3>
                <p>Name: <strong>{name || "-"}</strong></p>
                <p>Tickets: <strong>{tickets || "-"}</strong></p>
            </section>
        </main>
    );
}
export default Booking;
onChange={(e) => setName(e.target.value)}
onChange - Whenever the user changes inside the input box, it reflects in the booking details.
e is the event object. It contains information about what happened in the input.
e.target.value - Get the current value entered in the input box.
setName(...) - Take whatever the user typed and store it in name.
              const [name, setName] = useState("");
      
=> is an arrow function
State is data that can change while the application is running, and when state changes, React updates the UI.

async function handleBooking() {
        if (!name || !tickets || !stand || !venue ) {
            alert("Please fill all the details");
            return;
        }
        const { data, error } = await supabase
            .from("bookings")
            .insert([
            {
                name: name,
                tickets: Number(tickets),
                stand: stand,
                venue: venue
                
            }
            ]);
        if (error) {
            console.error(error);
            alert("Booking failed");
            return;
        }
        alert("Ticket booked successfully!");
        }
    }

Step 11  Create BookingHistory.jsx
src/pages/BookingHistory.jsx
import { useState } from "react";
import { supabase } from "../supabase";
function BookingHistory() {
  const [bookings, setBookings] = useState([]);
  async function getBookings() {
    const { data, error } = await supabase
      .from("bookings")
      .select("*");
    if (error) {
      console.error(error);
      alert("Failed to retrieve bookings");
      return;
    }
    setBookings(data);
  }
  return (
    <main className="history-page">
      <h1>Booking History</h1>
      <button className="history-button" onClick={getBookings}>
        View Bookings
      </button>
      <div className="history-table-wrapper">
        <table className="history-table">
          <thead>
            <tr>
              <th>Name</th>
              <th>Tickets</th>
              <th>Stand</th>
              <th>Venue</th>
            </tr>
          </thead>
          <tbody>
            {bookings.map((booking) => (
              <tr key={booking.id}>
                <td>{booking.name}</td>
                <td>{booking.tickets}</td>
                <td>{booking.stand}</td>
                <td>{booking.venue}</td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </main>
  );
}
export default BookingHistory;
